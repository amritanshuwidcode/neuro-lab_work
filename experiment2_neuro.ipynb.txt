        "#experiment-2\n",
        "\n",
        "import numpy as np\n",
        "import pandas as pd\n",
        "\n",
        "\n",
        "iq = np.array([120, 140, 130, 140], dtype=float)\n",
        "cgpa = np.array([8, 9, 7, 6], dtype=float)\n",
        "actual = np.array([7, 9, 5, 6], dtype=float)\n",
        "\n",
        "\n",
        "w1 = 0.01\n",
        "w2 = 0.10\n",
        "b = 0.0\n",
        "\n",
        "learning_rate = 0.0001\n",
        "\n",
        "rows = []\n",
        "\n",
        "for i in range(len(iq)):\n",
        "\n",
        "\n",
        "    prediction = w1 * iq[i] + w2 * cgpa[i] + b\n",
        "\n",
        "\n",
        "    error = prediction - actual[i]\n",
        "\n",
        "\n",
        "    loss = error ** 2\n",
        "\n",
        "\n",
        "    dw1 = 2 * error * iq[i]\n",
        "    dw2 = 2 * error * cgpa[i]\n",
        "    db = 2 * error\n",
        "\n",
        "\n",
        "    w1 = w1 - learning_rate * dw1\n",
        "    w2 = w2 - learning_rate * dw2\n",
        "    b = b - learning_rate * db\n",
        "\n",
        "    rows.append([\n",
        "        iq[i],\n",
        "        cgpa[i],\n",
        "        actual[i],\n",
        "        prediction,\n",
        "        loss,\n",
        "        w1,\n",
        "        w2,\n",
        "        b\n",
        "    ])\n",
        "\n",
        "df = pd.DataFrame(rows, columns=[\n",
        "    \"IQ\",\n",
        "    \"CGPA\",\n",
        "    \"Actual LPA\",\n",
        "    \"Predict\",\n",
        "    \"Loss\",\n",
        "    \"Updated w1\",\n",
        "    \"Updated w2\",\n",
        "    \"Updated b\"\n",
        "])\n",
        "\n",
        "print(df)"
